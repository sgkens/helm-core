# Power Only
# Batteries, reactor and solar on one panel; minimal binds.
@bind batts = group("Main Batteries", agg: avg)
@bind rx    = block("Reactor 1")
@bind solar = group("Solar Panels", agg: avg)

@screen PowerOnly
output: "Power LCD", 0
profile: standard
layout: grid(1, 1)
bg: #0a0d12
border_color: #eab308

@panel Grid at(0, 0, 1, 1) title: "POWER GRID" title_align: center
line: "BATTERY " bar(batts.percent) gradient(#ef4444, #eab308, #22c55e)
line: "CHARGE  " flow(batts.current) gradient(#22c55e, #16a34a)
line: "REACTOR " bar(rx.output) gradient(#22c55e, #eab308, #ef4444) unit("MW")
line: "SOLAR   " spark(solar.output) flat(#facc15)
