# Title Alignment
# Left, center and right titles with independent title colors and a border gradient.
@bind v = dynamic("demo.v", default: 58)

@screen Titles
output: "Titles LCD", 0
profile: auto
layout: grid(1, 3)
bg: #090b10
border_color: #475569

@panel Left at(0, 0, 1, 1) title: "LEFT TITLE" title_align: left title_color: #f43f5e border_color: #1e293b
line: "VALUE   " bar(v.val) gradient(#f43f5e, #fb7185)
@panel Center at(0, 1, 1, 1) title: "CENTER TITLE" title_align: center title_color: #22c55e border_color: #1e293b
line: "VALUE   " bar(v.val) gradient(#22c55e, #86efac)
@panel Right at(0, 2, 1, 1) title: "RIGHT TITLE" title_align: right title_color: #38bdf8 border_color: #1e293b
line: "VALUE   " bar(v.val) gradient(#38bdf8, #7dd3fc)
