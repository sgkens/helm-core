# Power Liquid Themes
# Animated power gauges on four independent screens; rename groups and LCDs to match your grid.
# Battery percent is stored charge. Generator percent is output divided by maximum output, not fuel or temperature.
@bind reactor = group("Reactors", agg: avg)
@bind battery = group("Batteries", agg: avg)
@bind solar = group("Solar Panels", agg: avg)
@bind engine = group("Hydrogen Engines", agg: avg)

@screen ReactorOutput
output: "Reactor Power LCD", 0
profile: auto
bg: #170d05
border_color: #f59e0b
@panel core at(0, 0, 1, 1) title: "REACTOR OUTPUT"
line: "PWR" tank(reactor.percent) gradient(#78350f, #f59e0b, #fef3c7)
